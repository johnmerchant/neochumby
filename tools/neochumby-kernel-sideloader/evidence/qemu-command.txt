# any incidental guest writes.  run-demo.py checks the original/scratch hashes.
# run-demo.py created WORK/nand-scratch.img from the original NAND, replacing
# only the PSP partition, then ran (paths abbreviated here; the script builds
# the same command):
qemu-system-arm -M chumby-classic \
  -kernel /Users/john/.local/share/neochumby-emu/run/zImage \
  -drive if=mtd,file=build/demo-final/nand-scratch.img,format=raw,snapshot=on \
  -device loader,file=build/payload.bin,addr=0xc2000000,force-raw=on \
  -snapshot -display none -no-reboot -monitor none \
  -serial unix:build/demo-final/serial.sock,server=on,wait=off \
  -append 'console=ttyS0,38400 root=/dev/mtdblock5 rootfstype=cramfs init=/linuxrc chumby_hwversion=3.8' \
  -icount shift=auto,sleep=off
